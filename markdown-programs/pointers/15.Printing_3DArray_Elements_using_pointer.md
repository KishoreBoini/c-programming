# 15.Printing_3DArray_Elements_using_pointer

**C source:** `c-programs/pointers/15.Printing_3DArray_Elements_using_pointer.c`

```c
#include<stdio.h>

void main()
{
	int Tmat[2][2][2]={{{1,2},{3,4}},
			{{6,7},{8,9}}
			};
	int (*ptr)[2][2];
	ptr=Tmat;
	for(int i=0;i<2;i++)
	{
		for(int j=0;j<2;j++)
		{
			for(int k=0;k<2;k++)
			{
				printf("%d ",ptr[i][j][k]);
			}
			printf("\n");
		}
	}
	printf("\n");
}
```

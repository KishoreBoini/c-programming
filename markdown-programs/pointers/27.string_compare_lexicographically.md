# String Compare Lexicographically

**C source:** `c-programs/pointers/27.string_compare_lexicographically.c`

## C Source Code

```c
#include<stdio.h>
#include<string.h>

int StringCompare(char *src,char *dest)
{
	while(*src!='\0' && *dest!='\0')
	{
		if(*src!=*dest)
		{
			if(*src>*dest)
				return 1;
			else
				return -1;
		}
		src++;
		dest++;
	}
	if(*dest)
		return -1;
	else if(*src)
		return 1;
return 0;
}

void main()
{
	char str1[20],str2[20];
	int ret;
	printf("Enter the string1 : %s",str1);
	scanf("%s",str1);
	printf("Enter the string2 : %s",str2);
	scanf("%s",str2);
	ret=StringCompare(str1,str2);
	if(ret==1)
		printf("given strings are not equal string1 is bigger\n");
	else if(ret==-1)
		printf("given strings are not equal string2 is bigger\n");
	else
		printf("given strings are equal\n");

}
```

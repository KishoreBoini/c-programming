# Matrix Multiplication Using Pointers

**C source:** `c-programs/pointers/39.Matrix_Multiplication_Using_Pointers.c`

## C Source Code

```c
#include<stdio.h>

void ProdMats(int row1,int col1,int row2,int col2,int (*ptr1)[col1],int (*ptr2)[col2],int (*ptr3)[col2])
{
	for(int i=0;i<row1;i++)
	{
		for(int j=0;j<col2;j++)
		{
			ptr3[i][j]=0;
			for(int k=0;k<col1;k++)
			{
				ptr3[i][j]+=ptr1[i][k]*ptr2[k][j];
			}
		}
	}
}

int main()
{
	int row1,row2,col1,col2;
	printf("Enter the row and column's of 1st matrix : ");
	scanf("%d%d",&row1,&col1);
	printf("Enter the row and column's of 2st matrix : ");
	scanf("%d%d",&row2,&col2);
	if(col1!=row2)
	{
		printf("Multiplication Not Possible \n");
		return -1;
	}
	int mat1[row1][col1];
	int mat2[row2][col2];
	int mat3[row1][col2];
	printf("Enter the First matrix elements : ");
	for(int i=0;i<row1;i++)
	{
		for(int j=0;j<col1;j++)
		{
			scanf("%d",&mat1[i][j]);
		}
	}
	printf("Enter the Second matrix elements : ");
	for(int i=0;i<row2;i++)
	{
		for(int j=0;j<col2;j++)
		{
			scanf("%d",&mat2[i][j]);
		}
	}
	ProdMats(row1,col1,row2,col2,mat1,mat2,mat3);
	printf("Product of the two Matrixes \n");
	for(int i=0;i<row1;i++)
	{
		for(int j=0;j<col2;j++)
		{
			printf("%d ",mat3[i][j]);
		}
		printf("\n");
	}
return 0;
}
```

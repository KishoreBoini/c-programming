# Number Of Vowels And Consonants In A String

**C source:** `c-programs/pointers/31.number_of_Vowels_and_Consonants_in_a_string.c`

## C Source Code

```c
#include<stdio.h>

void VowelsConsonants(char *ptr,int *vowels,int *consonants)
{
	while(*ptr!='\0')
	{
		if(*ptr == 'a'||*ptr == 'e' || *ptr == 'i' || *ptr == 'o' || *ptr == 'u' || *ptr == 'A'||*ptr == 'E' || *ptr == 'I' || *ptr == 'O' || *ptr == 'U')
		{
			(*vowels)++;
		}
		else
		{
			(*consonants)++;
		}
		ptr++;
	}
}

int main()
{
	char str[20];
	int vol=0,con=0;
	printf("Enter the String : ");
	scanf("%s",str);
	VowelsConsonants(str,&vol,&con);
	printf("Vowels = %d\nConsonants = %d\n",vol,con);

}
```

#include<stdio.h>

void  MatrixrightDiagonalsum(int n,int arr[n][n])
{
	int sum=0;
	for(int i=0;i<n;i++)
	{
		sum+=arr[i][n-i-1];
	}
	printf("The sum of the right diagonal of the given matrix is : %d\n`",sum);

}

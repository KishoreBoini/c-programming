#include<stdio.h>


void Next_Great_Element(int len,int arr[],int ge[])
{
	int k=0;
	int flag;
	for(int i=0;i<len;i++)
	{
		flag=0;
		for(int j=i+1;j<len;j++)
		{
			if(arr[j]>arr[i])
			{
				ge[k++]=arr[j];
				flag=1;
				break;	
			}
		}
		if(flag==0)
			ge[k++]=-1;


		
	}
}

int main()
{
	int n;
	printf("Enter the size of the Array : ");
	scanf("%d",&n);
	int arr[n];
	int nxt_ge[n];
	printf("Enter the Elements in the Array : ");
	for(int i=0;i<n;i++)
	{
		scanf("%d",&arr[i]);
	}
	Next_Great_Element(n,arr,nxt_ge);
	for(int i=0;i<n;i++)
	{
		printf("%d ",arr[i]);
	}
	printf("\n");
	for(int i=0;i<n;i++)
	{
		printf("%d ",nxt_ge[i]);
	}
	printf("\n");

return 0;
}

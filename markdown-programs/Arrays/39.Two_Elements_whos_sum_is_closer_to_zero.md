# 39.Two_Elements_whos_sum_is_closer_to_zero

**C source:** `c-programs/Arrays/39.Two_Elements_whos_sum_is_closer_to_zero.c`

```c
#include<stdio.h>
#include<limits.h>
#include<stdlib.h>

void Sum_Near_zero(int len,int arr[])
{
	int min_sum=INT_MAX;
	int num1=0,num2=0;
	for(int i=0;i<len-1;i++)
	{
		for(int j=i+1;j<len;j++)
		{
			int current_sum=arr[i]+arr[j];
			if(abs(current_sum)<abs(min_sum))
			{
				min_sum=current_sum;
				num1=arr[i];
				num2=arr[j];
			}
		}

	}
	printf("The two Elements who's sum is closer to zero are : %d and %d\n",num1,num2);
}

int main()
{
	int n;
	printf("Enter the size of the Array : ");
	scanf("%d",&n);
	int arr[n];
	printf("Enter the Elements in the Array : ");
	for(int i=0;i<n;i++)
	{
		scanf("%d",&arr[i]);
	}
	Sum_Near_zero(n,arr);

return 0;
}
```

# 48.Rversing_Array_Using_Recursion

**C source:** `c-programs/Arrays/48.Rversing_Array_Using_Recursion.c`

```c
#include<stdio.h>

void Reverse_Array(int arr[],int start,int end)
{
	if(start>=end)
		return;
	int temp=arr[start];
	arr[start]=arr[end];
	arr[end]=temp;
	Reverse_Array(arr,start+1,end-1);
}

int main()
{
	int n;
	printf("Enter the size of the Array : ");
	scanf("%d",&n);
	int arr[n];
	printf("Enter the Elements in the Array : ");
	for(int i=0;i<n;i++)
	{
		scanf("%d",&arr[i]);
	}
	Reverse_Array(arr,0,n-1);
	printf("The reversed Array : ");
	for(int i=0;i<n;i++)
	{
		printf("%d ",arr[i]);
	}
	printf("\n");

return 0;
}
```

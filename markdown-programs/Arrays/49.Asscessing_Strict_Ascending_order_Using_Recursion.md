#include<stdio.h>

int ifAscending(int arr[],int len,int n) 
{
	if(len==n-1)
		return 1;
	if(arr[len]>arr[len+1])
		return 0;
	
	ifAscending(arr,len+1,n);

}

int main()
{
	int n,re;
	printf("Enter the size of the Array : ");
	scanf("%d",&n);
	int arr[n];
	printf("Enter the Elements in the Array : ");
	for(int i=0;i<n;i++)
	{
		scanf("%d",&arr[i]);
	}
	re=ifAscending(arr,0,n);
	if(re)	
		printf("Strictly ascending \n");
	
	else
		printf("Not Strictly Ascending \n");
return 0;
}

# 34.Reverse_the_Elements_of_Array

**C source:** `c-programs/Arrays/34.Reverse_the_Elements_of_Array.c`

```c
#include<stdio.h>

void Reverse_Array(int len,int arr[])
{
	for(int i=0,j=len-1;i<j;i++,j--)
	{
		int temp=arr[i];
		arr[i]=arr[j];
		arr[j]=temp;
	}

}

int main()
{
	int n;
	printf("Enter the size of Array : ");
	scanf("%d",&n);
	int arr[n];
	printf("Enter the Elements in the Array : ");
	for(int i=0;i<n;i++)
	{
		scanf("%d",&arr[i]);
	}
	Reverse_Array(n,arr);
	printf("Reversed array : \n");
	for(int i=0;i<n;i++)
	{
		printf("%d ",arr[i]);
	}
	printf("\n");

return 0;		
}
```

# 43.Matrix_symetric_or_not

**C source:** `c-programs/Arrays/43.Matrix_symetric_or_not.c`

```c
#include<stdio.h>

int ifSymmetry(int row,int col,int mat[row][col])
{
	int count=0;
	for(int i=0;i<row;i++)
	{
		for(int j=0;j<col;j++)
		{
			if(mat[i][j]==mat[j][i])
			{
				count++;
			}
		}
	}
	if(count>=row*col)
	{
		return 1;
	}
return 0;
}

int main()
{
	int r,c,re;
	printf("Enter the rows of the Array : ");
	scanf("%d",&r);
	printf("Enter the columns of the Array : ");
	scanf("%d",&c);
	if(r!=c)
	{
		printf("Not a symmetric Matrix!!\n");
		return 0;
	}
	int mat[r][c];
	printf("Enter the Elements in the Matrix : ");
	for(int i=0;i<r;i++)
	{
		for(int j=0;j<c;j++)
		{
			scanf("%d",&mat[i][j]);
		}
	}
	re=ifSymmetry(r,c,mat);
	if(re)
	{
		printf("Symmetric Matrix!\n");
	}
	else
	{
		printf("Not a symmetric Matrix!!\n");
	}

return 0;
}
```

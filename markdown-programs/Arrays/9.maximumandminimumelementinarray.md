#include<stdio.h>

void MaxMinofarray(int arr[],int len)
{
	int temp1=arr[0],temp2=arr[0];
	for(int i=0;i<len;i++)
	{
		if(arr[i]>temp1)
		{
			temp1=arr[i];
		}
		if(arr[i]<temp2)
		{
			temp2=arr[i];
		}
	}
	printf("max=%d min=%d\n",temp1,temp2);

}

int main()
{
	int n;
	printf("Enter the n value : ");
	scanf("%d",&n);
	int arr[n];
	for(int i=0;i<n;i++)
	{
		scanf("%d",&arr[i]);
	}
	MaxMinofarray(arr,n);
return 0;
}

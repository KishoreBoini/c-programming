#include<stdio.h>
								 	
void Sprial_Display(int r,int c,int arr[r][c])
{
	int top = 0;
	int bottom = r-1;
	int right = c-1;
	int left = 0;
	printf("Sprial Printf Of Given Matrix: \n");
	while((top<=bottom) && (left<=right))
	{
		for(int i=left;i<=right;i++)
			printf("%d ",arr[top][i]);
		top=top+1;
		for(int j=top;j<=bottom;j++)
			printf("%d ",arr[j][right]);
		right=right-1;
		if(top<=bottom)
		{
			for(int k=right;k>=left;k--)
			{
				printf("%d ",arr[bottom][k]);
			}
			bottom=bottom-1;
		}
		if(left<=right)
		{
			for(int l=bottom;l>=top;l--)
			{
				printf("%d ",arr[l][left]);
			}
			left=left+1;
		}
	}
	printf("\n");
}

int main()
{
	int r,c;
	printf("Enter the rows of the matrix : ");
	scanf("%d",&r);
	printf("Enter the clomuns of the matrix : ");
	scanf("%d",&c);
	int arr[r][c];
	printf("Enter the Elements of the matrix : ");
	for(int i=0;i<r;i++)
	{
		for(int j=0;j<c;j++)
		{
			scanf("%d",&arr[i][j]);
		}
	}
	Sprial_Display(r,c,arr);
return 0;
}

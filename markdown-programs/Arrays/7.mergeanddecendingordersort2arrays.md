#include<stdio.h>

void mergearrays(int src[],int dest[],int res[],int n);
void sortarray(int arr[],int n);

void mergearrays(int src[],int dest[],int res[],int n)
{
	int i;
	for(i=0;i<n;i++)
	{
		res[i]=src[i];
	}
	for(int j=0;j<n;j++)
	{
		res[i]=dest[j];
		i++;
	}

	sortarray(res,i);

}
void sortarray(int arr[],int n)
{
	int i;
	for(i=0;i<n;i++)
	{
		for(int j=i+1;j<n;j++)
		{

			if(arr[i]>arr[j])
			{
				int temp=arr[i];
				arr[i]=arr[j];
				arr[j]=temp;
			}
		}
	}
	 uniqueelements(arr,i);
	printf("Merged and sorted array !\n");
	for(int i=0;i<n;i++)
	{
		printf("%d ",arr[i]);
	}
	printf("\n");
}

{
	int j=0;
	for(int i=1;i<n;i++)
	{
		if(darr[i]!=darr[j])		//[1,2,3,1,4,5,5]
		{
			j++;
			darr[j]=darr[i];
			

		}
	}
	printf("unique elements in the array \n");
	for(int i=0;i<j+1;i++)
	{
		printf("%d ",darr[i]);

}

# Distinct Array Or Not

**C source:** `c-programs/Arrays/44.Distinct_Array_or_Not.c`

## C Source Code

```c
#include<stdio.h>

void Distinct_Array(int len,int arr[])
{
	int max=arr[0];
	for(int i=1;i<len;i++)
	{
		if(arr[i]>max)
		{
			max=arr[i];
		}
	}
	int freq[max+1],flag=0;
	for(int i=0;i<len;i++)
	{
		freq[i]=0;
	}
	
	for(int i=0;i<len;i++)
	{
		freq[arr[i]]++;
	}
	for(int i=0;i<len;i++)
	{
		if(freq[i]>1)
		{
			flag=1;
			break;
		}
	}
	if(flag)
		printf("Non distinct Array\n");

	else
		printf("Distinct Array\n");

}

int main()
{
	int n;
	printf("Enter the size of the Array : ");
	scanf("%d",&n);
	int arr[n];
	printf("Enter the Elements in the Array : ");
	for(int i=0;i<n;i++)
	{
		scanf("%d",&arr[i]);
	}
	Distinct_Array(n,arr);
	
return 0;
}
```

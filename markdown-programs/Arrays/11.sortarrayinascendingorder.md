#include<stdio.h>

int* sortAscending(int* unsor,int len)
{
	int a;								
	for(int i=0;i<len;i++)
	{
		for(int j=i+1;j<len;j++)
		{
			if(unsor[i]>unsor[j])
			{
				int temp=unsor[j];
				unsor[j]=unsor[i];
				unsor[i]=temp;
			}
		}
	}
	return unsor;

}

int main()
{
	int n;
	int *sortedarray;
	printf("Enter the size of array : ");
	scanf("%d",&n);
	int arr[n];
	printf("Enter the array elements : ");
	for(int i=0;i<n;i++)
	{
		scanf("%d",&arr[i]);

	}
	sortedarray = sortAscending(arr,n);
	printf("sorted array (Acsending oerder) : \n");
	for(int i=0;i<n;i++)
	{
		printf("%d ",sortedarray[i]);
		
	}
	printf("\n");
return 0;
}

# Complex Number Sum

**C source:** `c-programs/structures/5.Complex_number_Sum.c`

## C Source Code

```c
#include<stdio.h>

struct complexNum{
	float real;
	float img;
};

struct complexNum Add_complex_num(struct complexNum n1,struct complexNum n2)
{
	struct complexNum add;
	add.real=n1.real+n2.real;
	add.img=n1.img+n2.img;
	return add;
}

int main()
{
	struct complexNum n1,n2,sum;
	printf("Enter the first complex number : (xx+xxi) ");
	scanf("%f+%fi",&n1.real,&n1.img);
	printf("Enter the second complex number : (xx+xxi) ");
	scanf("%f+%fi",&n2.real,&n2.img);
	sum = Add_complex_num(n1,n2);
	printf("sum = %0.1f+%.1fi\n",sum.real,sum.img);
}
```

# Book Details

**C source:** `c-programs/structures/3.Book_Details.c`

## C Source Code

```c
#include<stdio.h>

struct Book{
	char title[40];
	char author[20];
	long int ISBM;
	int pages;
};

struct Book  Store_data()
{
	struct Book details;
	printf("Enter the Book Title : ");
	scanf("%s",details.title);
	printf("\nEnter the Author Name : ");
	scanf("%s",details.author);
	printf("\nEnter the Book Code : ");
	scanf("%ld",&details.ISBM);
	printf("\nEnter the No.of Pages :");
	scanf("%d",&details.pages);

	return details;

}

int main()
{
	struct Book d;
	d=Store_data();
	printf("\nbook title = %s\n",d.title);
	printf("book Author = %s\n",d.author);
	printf("book code = %ld\n",d.ISBM);
	printf("book pages = %d\n",d.pages);
return 0;
}
```

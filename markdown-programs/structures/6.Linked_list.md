#include<stdio.h>
#include <stdlib.h>

struct node{
	int data;
	struct node *nxt;
};

struct node *pHead ;


struct node* Create_node(int num)
{
	struct node *pNew,*pTrave;
	pNew=malloc(sizeof(struct node));

	if(pHead==NULL)
	{
		pHead=pNew;
		pHead->data=num;
		pHead->nxt=NULL;
		return pHead;
	}
	pTrave=pHead;
	while(pTrave->nxt!=NULL)
	{
		pTrave=pTrave->nxt;

	}
	pTrave->nxt=pNew;
	pNew->data=num;
	pNew->nxt=NULL;

return pHead;
}

int main()
{
	Create_node(10);
	Create_node(20);
	Create_node(30);
	Create_node(40);
	Create_node(50);
	Create_node(60);
	Create_node(70);
	Create_node(80);

return 0;
}

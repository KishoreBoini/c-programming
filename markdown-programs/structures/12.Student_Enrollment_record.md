# Student Enrollment Record

## C Source Code

```c
#include<stdio.h>

struct enrollment{
	long int studentID;
	char courseName[30];
	char grade;
};

void Student_Enrollment(struct enrollment data[],char option,long int ID)
{
	static int index=0;
	char more;
	int flag=0;
	if(option=='a')
	{
		do
		{
			printf("Enter the Student ID : ");
			scanf("%ld",&data[index].studentID);
			printf("\nEnter the course name you enrolled into : ");
			scanf("%s",data[index].courseName);
			printf("\nEnter your grade : ");
			scanf(" %c",&data[index].grade);
			index++;
			printf("Wnat to add more ID's (y/n) : ");
			scanf(" %c",&more);
		}while(more=='y'||more=='Y');
	}

	else if(option=='b')
	{
		for(int i=0;i<index;i++)
		{
			if(data[i].studentID==ID)
			{
				printf("**ID found!**\n");
				printf("Student ID : %ld\n",data[i].studentID);
				printf("Course Name : %s\n",data[i].courseName);
				printf("Grade obtained : %c\n",data[i].grade);
				flag=1;
					break;
			}
		}
		if (flag==0)
		{
			printf("ID not found!!\n");
		}
		printf("\n\n\n");
		
	}
		
}

int main()
{
	char option;
	long int search_ID;
	struct enrollment student[20];
	while(1)
	{
		printf(" **Student Enrollment**\n   **Options**   \na.New Enrollment\nb.search for ID\nq.quit!!\n");
		scanf(" %c",&option);
		if(option=='b')
		{
			printf("Enter the search ID : ");
			scanf("%ld",&search_ID);
		}
		else
		{
			search_ID=0;
		}
		Student_Enrollment(student,option,search_ID);
		if(option=='q')
			break;
	}
	
return 0;
}
```

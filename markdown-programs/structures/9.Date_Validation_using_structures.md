struct date{
	int day;
	int month;
	int year;
	
};


int isdateValid(struct date d)
{
	int isleapyear=0;

	if(d.day<1 || d.day>31)
	{
		return 1;
	}
	if(d.month<1 || d.month>12)
	{
		return 1;
	}
	if(d.year<1000 || d.year>3000)
	{
		return 1;
	}
	if((d.year%100!=0 && d.year%4==0) || d.year%400==0)
	{
		 isleapyear=1;
	}
	
	if(d.month == 1 || d.month==3 || d.month==5 || d.month==7 || d.month==8 ||d.month== 10 || d.month==12)
	{
		if(d.day>=1 && d.day <=31)
		{
			return 0;
		}
		else
			return 1;
	}
	if(d.month == 4 || d.month==6 || d.month==9 || d.month==11 )
	{
		if(d.day>=1 && d.day<=30)
		{
			return 0;
		}
		else
			return 1;
	}
	if(d.month == 2)
	{
		if(isleapyear)
		{
			if(d.day>=1 && d.day<=29)
			{
				return 0;			
			}
			else 
				return 1;

		}
		else
			if(d.day>=1 && d.day<=28)
			{
				return 0;
			}
			else
				return 1;
	}

return 1;
}

int main()

return 0;

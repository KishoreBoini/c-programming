//1. Write a function to count the number of occurrences of an element in a single linked list.

#include<stdio.h>
#include <stdlib.h>

struct node{
	int val;
	struct node *nxt;
};

struct node *pHead;

void createNode(int data)
{
	struct node *pNew,*pTrave;
	pNew=malloc(sizeof(struct node));
	if(pHead==NULL)
	{
		pHead=pNew;
		pHead->val=data;
		pHead->nxt=NULL;
		return;
	}
	pTrave=pHead;
	while(pTrave->nxt!=NULL)
	{
		pTrave=pTrave->nxt;
	}
	pNew->val=data;
	pNew->nxt=NULL;
	pTrave->nxt=pNew;
return;
}

void Display()
{
	printf("Dispaly\n");
	int i=1;
	struct node *pTrave;
	pTrave=pHead;
	while(pTrave!=NULL)
	{
		printf("%d node's data = %d\n",i,pTrave->val);
		i++;
		pTrave=pTrave->nxt;
	}
return;		
}

void frequency()
{
	printf("using nested loops\n");
	int count,flag;
	struct node *pTrave,*Pnode,*prev;
	Pnode=pHead;
	while(Pnode!=NULL)
	{
		prev=pHead;
		flag=0;
		while(prev!=Pnode)
		{
			if(Pnode->val==prev->val)
			{
				flag=1;
			}
			prev=prev->nxt;
		}

			count=0;
			pTrave=pHead;
			while(pTrave!=NULL)
			{
				if(flag)
					break;
				if(Pnode->val==pTrave->val)
				{
					count++;
				}
				pTrave=pTrave->nxt;
			}
			if(flag==0)
				printf("%d's occurance is %d times \n",Pnode->val,count);
			
		Pnode=Pnode->nxt;
	}
	//using frequency array 
	Pnode=pHead;
	printf("Uisng Frequency Array\n");
	int arr[100],freq[100]={0},visited[100]={0},i=0;
	while(Pnode!=NULL)
	{
		arr[i]=Pnode->val;
		i++;
		Pnode=Pnode->nxt;
	}

	for(int j=0;j<i;j++)
	{
		freq[arr[j]]++;
	}
	for(int k=0;k<i;k++)
	{
		if(visited[arr[k]]==0)
		{
			printf("%d times %d\n",arr[k],freq[arr[k]]);
			visited[arr[k]]=1;
		}
	}


		
	
return;
}


int main()
{

return 0;

//8. Write recursive functions for the following operations on a linked list.
//(Viii) Search for an element in the list

#include<stdio.h>
#include <stdlib.h>

struct node{
	int data;
	struct node *nxt;
};

struct node *pHead;

void createNode(int val)
{
	struct node *pNew,*pTrave;
	pNew=malloc(sizeof(struct node));
	if(pHead==NULL)
	{
		pHead=pNew;
		pHead->data=val;
		pHead->nxt=NULL;
		return;
	}
	pTrave=pHead;
	while(pTrave->nxt!=NULL)
	{
		pTrave=pTrave->nxt;
	}
	pNew->data=val;
	pNew->nxt=NULL;
	pTrave->nxt=pNew;
return;
}


struct node* find(struct node *add,int val)
{
	if(add->data==val)
		return add;

	add->nxt=find(add->nxt,val);
}

int main()
{
	struct node *res;
	createNode(10);
	createNode(20);
	createNode(30);
	createNode(20);
	createNode(50);
	createNode(10);
	res=find(pHead,30);
	printf("%d",res->data);
	
return 0;
}

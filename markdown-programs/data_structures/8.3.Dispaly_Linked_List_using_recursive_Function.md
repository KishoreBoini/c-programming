//Write recursive functions for the following operations on a linked list
//Display the linked list

#include<stdio.h>
#include<stdlib.h>

struct node{
	int data;
	struct node *nxt;
};

struct node *pHead;

void createNode(int val)
{
	struct node *pNew;
	pNew=malloc(sizeof(struct node));

	if(pHead==NULL)
	{
		pHead=pNew;
		pHead->data=val;
		pHead->nxt=NULL;
		return;
	}

	pNew->nxt=pHead;
	pHead=pNew;
	pHead->data=val;
return;
}

void DisplayR(struct node *add,int count)
{
	if(add==NULL)
		return;

	printf("data in %d mode is %d\n",count,add->data);
	count++;
	add=add->nxt;
	DisplayR(add,count);
}

int main()
{
	for(int i=10;i<100;i+=10)
	{
		createNode(i);
	}
	DisplayR(pHead,0);
}

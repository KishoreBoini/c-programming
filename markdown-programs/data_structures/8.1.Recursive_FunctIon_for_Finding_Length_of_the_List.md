//Write recursive functions for the following operations on a linked list.
//(i) Find length of the list


#include<stdio.h>
#include<stdlib.h>

struct node{
	int data;
	struct node *nxt;
};

struct node *pHead;

struct node* createNode(int val)
{
	struct node *pNew,*pTrave;
	pNew=malloc(sizeof(struct node));

	if(pHead==NULL)
	{
		pHead=pNew;
		pHead->data=val;
		pHead->nxt=NULL;
		return pHead;
	}

	pTrave=pHead;
	while(pTrave->nxt!=NULL)
	{
		pTrave=pTrave->nxt;
	}
	pTrave->nxt=pNew;
	pNew->data=val;
	pNew->nxt=NULL;

return pHead;
}

void LengthofListR(struct node *add,int *count)
{
	if(add==NULL)
		return;

	(*count)+=1;
	add=add->nxt;
	LengthofListR(add,count);
}
